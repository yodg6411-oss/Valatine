    <script>
        // Liste des messages et des images humoristiques qui changent à chaque tentative de fuite
        const etapes = [
            { text: "T'as cliqué non par accident 🥲", img: "https://media.giphy.com/media/3o7TKSjRrfIPjeiVyM/giphy.gif" },
            { text: "T'es genre vraiment vraiment sûr·e ?? 🤨", img: "https://media.giphy.com/media/12bjQ7uajesAtC/giphy.gif" },
            { text: "S'il te plaît pookie ? 🥺👉👈", img: "https://media.giphy.com/media/1HRvF5PBdIJ77bIPUs/giphy.gif" },
            { text: "Si tu dis non je vais être trop triste 😭", img: "https://media.giphy.com/media/7SF5scGB2AFrgsXP63/giphy.gif" },
            { text: "Relis la question une dernière fois ? Dernière chance... 🤡", img: "https://media.giphy.com/media/efSePxL3C27zLG0mHY/giphy.gif" }
        ];

        let indexEtape = 0;

        function deplacerNon() {
            const btnNon = document.getElementById('btn-non');
            const btnOui = document.getElementById('btn-oui');
            const questionText = document.getElementById('question-text');
            const memeImg = document.getElementById('mame-img');

            // Faire bouger le bouton "Non" aléatoirement dans la page
            const x = Math.random() * (window.innerWidth - 150);
            const y = Math.random() * (window.innerHeight - 100);
            
            btnNon.style.left = `${x}px`;
            btnNon.style.top = `${y}px`;

            // Faire grandir un tout petit peu le bouton "OUI" pour inciter à cliquer dessus
            let currentScale = 1 + (indexEtape * 0.1);
            btnOui.style.transform = `scale(${currentScale})`;

            // Changer le texte et l'image en boucle
            if (indexEtape < etapes.length) {
                questionText.innerText = etapes[indexEtape].text;
                memeImg.src = etapes[indexEtape].img;
                indexEtape++;
            } else {
                indexEtape = 0; // Recommence la série si on insiste trop
            }
        }

        function clicOui() {
            document.getElementById('main-card').style.display = 'none';
            document.getElementById('message-final').style.display = 'block';
            document.body.style.backgroundColor = '#ffb3c6';
        }
    </script>
